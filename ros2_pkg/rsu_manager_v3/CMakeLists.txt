cmake_minimum_required(VERSION 3.8)
project(rsu_manager_v3)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(roa_interfaces REQUIRED)
find_package(rsu_manager_v2 REQUIRED)
find_package(realtime_tools REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(Threads REQUIRED)
add_executable(rsu_manager_v3_node src/rsu_manager_v3_node.cpp)
target_include_directories(rsu_manager_v3_node PRIVATE include)
target_link_libraries(rsu_manager_v3_node rsu_manager_v2::rsu_manager_v2_core Threads::Threads)
ament_target_dependencies(rsu_manager_v3_node rclcpp roa_interfaces rsu_manager_v2 realtime_tools ament_index_cpp)
install(TARGETS rsu_manager_v3_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_torque_mapper test/test_torque_mapper.cpp)
  target_include_directories(test_torque_mapper PRIVATE include)
  target_link_libraries(test_torque_mapper rsu_manager_v2::rsu_manager_v2_core)
  ament_target_dependencies(test_torque_mapper rsu_manager_v2)
  target_compile_definitions(test_torque_mapper PRIVATE RSU_TEST_LUT="${rsu_manager_v2_DIR}/../data/rsu_lut_v2.bin")
endif()
ament_package()
